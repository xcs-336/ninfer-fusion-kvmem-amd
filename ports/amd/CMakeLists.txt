cmake_minimum_required(VERSION 3.25)
project(ninfer_amd_qualification LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_SCAN_FOR_MODULES OFF)
set(ROCM_ROOT "D:/driver/ROCm/10.0.0" CACHE PATH "Native Windows AMD ROCm SDK")
if(NOT CMAKE_CXX_COMPILER_ID MATCHES "Clang")
  message(FATAL_ERROR "Use the installed AMD clang++ compiler")
endif()
if(NOT EXISTS "${ROCM_ROOT}/include/hip/hip_runtime.h")
  message(FATAL_ERROR "ROCM_ROOT must contain the AMD HIP SDK")
endif()
add_executable(ninfer_amd_mma gfx12-mma/test_mma.hip)
set_source_files_properties(gfx12-mma/test_mma.hip PROPERTIES LANGUAGE CXX)
target_include_directories(ninfer_amd_mma PRIVATE "${ROCM_ROOT}/include")
target_compile_definitions(ninfer_amd_mma PRIVATE __HIP_PLATFORM_AMD__=1)
target_compile_options(ninfer_amd_mma PRIVATE -x hip -O2 -fno-fast-math -ffp-contract=off
  --offload-arch=gfx1201 -mno-wavefrontsize64 "--rocm-path=${ROCM_ROOT}"
  "--rocm-device-lib-path=${ROCM_ROOT}/lib/llvm/amdgcn/bitcode")
target_link_options(ninfer_amd_mma PRIVATE --hip-link --offload-arch=gfx1201 "--rocm-path=${ROCM_ROOT}")
target_link_libraries(ninfer_amd_mma PRIVATE "${ROCM_ROOT}/lib/amdhip64.lib")
enable_testing()
include(GpuTests.cmake)
include(PrimitiveTests.cmake)
ninfer_add_gpu_test(native_bf16_mma ninfer_amd_mma 30)
option(NINFER_AMD_VMM_DIAGNOSTICS "Build known-failing native VMM driver diagnostics" OFF)
if(NINFER_AMD_VMM_DIAGNOSTICS)
add_executable(ninfer_amd_vmm_isolation tests/vmm_isolation.hip)
set_source_files_properties(tests/vmm_isolation.hip PROPERTIES LANGUAGE CXX)
target_include_directories(ninfer_amd_vmm_isolation PRIVATE "${ROCM_ROOT}/include")
target_compile_definitions(ninfer_amd_vmm_isolation PRIVATE __HIP_PLATFORM_AMD__=1)
target_compile_options(ninfer_amd_vmm_isolation PRIVATE -x hip --offload-arch=gfx1201
  "--rocm-path=${ROCM_ROOT}" "--rocm-device-lib-path=${ROCM_ROOT}/lib/llvm/amdgcn/bitcode")
target_link_options(ninfer_amd_vmm_isolation PRIVATE --hip-link --offload-arch=gfx1201 "--rocm-path=${ROCM_ROOT}")
target_link_libraries(ninfer_amd_vmm_isolation PRIVATE "${ROCM_ROOT}/lib/amdhip64.lib")
ninfer_add_gpu_test(native_vmm_isolation ninfer_amd_vmm_isolation 30)
ninfer_add_gpu_test(native_vmm_isolation_warm ninfer_amd_vmm_isolation 30 65536 warm)
ninfer_add_gpu_test(native_vmm_isolation_2m ninfer_amd_vmm_isolation 30 2097152 warm)
ninfer_add_gpu_test(native_vmm_isolation_split ninfer_amd_vmm_isolation 30 2097152 warm split)
endif()
option(NINFER_AMD_CORE "Build and qualify the optimized baseline's actual Core" OFF)
if(NINFER_AMD_CORE)
  include(Core.cmake)
endif()
